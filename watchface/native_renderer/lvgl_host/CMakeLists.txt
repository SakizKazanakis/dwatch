cmake_minimum_required(VERSION 3.16)
project(monolith_lvgl8_host C)
set(CMAKE_C_STANDARD 11)
if(NOT MONOLITH_LVGL_SOURCE)
  message(FATAL_ERROR "Pass -DMONOLITH_LVGL_SOURCE=/path/to/lvgl-v8.3")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "LVGL host config" FORCE)
add_subdirectory("${MONOLITH_LVGL_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
add_library(monolith_lvgl8_adapter STATIC ../src/monolith_lvgl8_adapter.c)
target_include_directories(monolith_lvgl8_adapter PUBLIC ../include)
target_link_libraries(monolith_lvgl8_adapter PUBLIC lvgl)
add_executable(monolith_lvgl8_host_test test_lvgl8_host.c)
target_include_directories(monolith_lvgl8_host_test PRIVATE ../include)
target_link_libraries(monolith_lvgl8_host_test PRIVATE monolith_lvgl8_adapter lvgl)
