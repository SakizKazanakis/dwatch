cmake_minimum_required(VERSION 3.16)
project(monolith_lvgl7_host C)
set(CMAKE_C_STANDARD 11)
if(NOT MONOLITH_LVGL_SOURCE)
  message(FATAL_ERROR "Pass -DMONOLITH_LVGL_SOURCE=/path/to/public/lvgl-v7")
endif()
add_subdirectory("${MONOLITH_LVGL_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
target_include_directories(lvgl PUBLIC "${MONOLITH_LVGL_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_library(monolith_lvgl7_adapter STATIC ../src/monolith_lvgl7_adapter.c)
target_include_directories(monolith_lvgl7_adapter PUBLIC ../include)
target_link_libraries(monolith_lvgl7_adapter PUBLIC lvgl)
add_executable(monolith_lvgl7_host_test test_lvgl7_host.c)
target_include_directories(monolith_lvgl7_host_test PRIVATE ../include)
target_link_libraries(monolith_lvgl7_host_test PRIVATE monolith_lvgl7_adapter lvgl)
