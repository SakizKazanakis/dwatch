cmake_minimum_required(VERSION 3.16)
project(monolith_native_renderer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(MONOLITH_SANITIZERS "ASan and UBSan host tests" OFF)
add_library(monolith_renderer STATIC src/monolith_ezip_native.c src/monolith_framebuffer.c)
target_include_directories(monolith_renderer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(monolith_renderer PRIVATE -Wall -Wextra -Werror)
if(MONOLITH_SANITIZERS)
 target_compile_options(monolith_renderer PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(monolith_renderer PUBLIC -fsanitize=address,undefined)
endif()
add_executable(monolith_decode_to_ppm tests/decode_to_ppm.c)
target_link_libraries(monolith_decode_to_ppm PRIVATE monolith_renderer)
add_executable(monolith_renderer_stress tests/test_renderer_stress.c)
target_link_libraries(monolith_renderer_stress PRIVATE monolith_renderer)
enable_testing()
set(MONOLITH_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/../../overlay/app/src/main/assets/monolith/clock_bg.bin")
if(EXISTS "${MONOLITH_FIXTURE}")
 add_test(NAME native_ezip_decode COMMAND monolith_decode_to_ppm "${MONOLITH_FIXTURE}" "${CMAKE_CURRENT_BINARY_DIR}/decoded.ppm")
 add_test(NAME renderer_1000_cycles COMMAND monolith_renderer_stress "${MONOLITH_FIXTURE}")
endif()
